# Phase 0 task contract

- `P0-01`â€“`P0-09` preserve the nine categories in the original Phase 0 plan. `P0-10` adds a distinct out-of-scope case; it is not a renumbering.
- IDs are stable. Materially changed task semantics require a new ID.
- `tasks.json` contains agent-visible request/context/constraints and public acceptance-check identifiers, but not `expected_behavior`. `evaluator.json` contains evaluator-only expected behavior and acceptance checks. The runner must never expose evaluator metadata in the prompt. File separation reduces accidental leakage but is not a security boundary by itself.
- `fixture:p0-NN-v1` values are named placeholders, not existing Git commits. Replace them with real commit hashes or immutable fixture IDs when reset states exist.
- The manifest describes intended tasks; task-specific buggy revisions, hidden tests, reset orchestration, patch evaluation, and run logging are not implemented yet.
- Clarification/rejection tasks may correctly produce no patch; evaluate their response and absence of unauthorized edits instead of requiring code tests on a nonexistent patch.

## Clean base behavior
- Discount: 0% below 100; 5% from 100 to below 500; 10% at 500+.
- Pricing rejects negative unit prices and quantities and delegates to discount calculation.
- Inventory subtracts reservations, clamps sellable quantity at zero, and rejects negative values.
- Currency formatting requires a three-letter alphabetic code and formats to two decimals.

$ErrorActionPreference = "Stop"
###What it does:
# Path Mapping: It dynamically locates the main repository (mock_python_repo), the task manifest (fixtures\tasks.json), and the isolated sibling worktree (p0-01-evaluation).
##Authoritative Read: It reads tasks.json, parses the JSON data, and extracts the base_revision specifically assigned to task P0-01.
###Integrity Validation: It enforces a strict rule that the target commit hash must be a valid, full 40-character Git commit hash and verifies via git cat-file that the commit actually exists in your Git database before attempting any operations.
# Main repository: mock_python_repo
$MainRepoRoot = Split-Path -Parent $PSScriptRoot
$ManifestPath = Join-Path $MainRepoRoot "fixtures\tasks.json"

# Evaluation worktree is a sibling of mock_python_repo.
$WorktreePath = Join-Path (Split-Path -Parent $MainRepoRoot) "p0-01-evaluation"

function Invoke-Git {
    param(
        [string]$WorkingDirectory,
        [string[]]$GitArguments
    )

    $output = & git -C $WorkingDirectory @GitArguments
    if ($LASTEXITCODE -ne 0) {
        throw "Git command failed: git -C `"$WorkingDirectory`" $($GitArguments -join ' ')"
    }

    return $output
}

# Read the authoritative P0-01 revision from the manifest.
if (-not (Test-Path $ManifestPath)) {
    throw "Task manifest not found: $ManifestPath"
}

$manifest = Get-Content $ManifestPath -Raw | ConvertFrom-Json
$task = @($manifest.tasks | Where-Object { $_.task_id -eq "P0-01" })

if ($task.Count -ne 1) {
    throw "Expected exactly one P0-01 entry in tasks.json."
}

$TargetCommit = [string]$task[0].base_revision

if ($TargetCommit -notmatch '^[0-9a-fA-F]{40}$') {
    throw "P0-01 base_revision must be a full 40-character Git commit hash."
}

# Confirm the recorded revision exists in the main repository.
Invoke-Git $MainRepoRoot @("cat-file", "-e", "$TargetCommit^{commit}") |
    Out-Null

# Do not create or overwrite a worktree automatically in this first version.
if (-not (Test-Path $WorktreePath -PathType Container)) {
    throw "Evaluation worktree not found: $WorktreePath. Create it with git worktree add first."
}

$ActualRoot = (Invoke-Git $WorktreePath @("rev-parse", "--show-toplevel") |
    Select-Object -First 1).Trim()

$ActualRoot = [System.IO.Path]::GetFullPath(
    $ActualRoot.Replace('/', '\')
).TrimEnd('\')

$ExpectedRoot = [System.IO.Path]::GetFullPath(
    $WorktreePath
).TrimEnd('\')

if (-not $ActualRoot.Equals(
    $ExpectedRoot,
    [System.StringComparison]::OrdinalIgnoreCase
)) {
    throw "The target directory is not the expected evaluation worktree. No reset performed."
}
# Safety check: refuse to destroy tracked, staged, or untracked changes.
$status = Invoke-Git $WorktreePath @(
    "status", "--porcelain", "--untracked-files=all"
)

if ($status.Count -gt 0 -and ($status -join "").Trim().Length -gt 0) {
    Write-Host "RESET REFUSED: evaluation worktree has changes:" -ForegroundColor Yellow
    $status | ForEach-Object { Write-Host $_ }
    throw "Commit, back up, or deliberately remove those changes before retrying."
}

# Safe to reset only after confirming the worktree is clean.
Write-Host "Resetting P0-01 evaluation worktree to $TargetCommit"
Invoke-Git $WorktreePath @("reset", "--hard", $TargetCommit) |
    ForEach-Object { Write-Host $_ }

$FinalCommit = (Invoke-Git $WorktreePath @("rev-parse", "HEAD") |
    Select-Object -First 1).Trim()

if ($FinalCommit -ne $TargetCommit) {
    throw "Verification failed: HEAD does not match the P0-01 target revision."
}

$finalStatus = Invoke-Git $WorktreePath @(
    "status", "--porcelain", "--untracked-files=all"
)

if ($finalStatus.Count -gt 0 -and ($finalStatus -join "").Trim().Length -gt 0) {
    throw "Verification failed: evaluation worktree is not clean after reset."
}

Write-Host ""
Write-Host "P0-01 reset: PASS" -ForegroundColor Green
Write-Host "Revision: $FinalCommit"
Write-Host "Worktree: $WorktreePath"
Write-Host "Working tree: clean"

$ErrorActionPreference = "Stop"

$RunId = [guid]::NewGuid().ToString("N")
$Result = [ordered]@{

    run_id = $RunId
    task_id = "P0-01"

    patch_path = $null

    full_suite_baseline = $null
    patch_applies = $false
    patch_check_output = $null
    tests_after_patch = $null
    changed_paths = @()
    protected_paths_changed = @()
    scope_compliant = $false
    accepted = $false
    diagnostics = [ordered]@{
        baseline_tests_passed = $null
        candidate_tests_passed = $null
        patch_check_passed = $false
        scope_compliant = $false
        full_suite_baseline_passed = $null
    }
    error = $null
    note = "Local first-pass evaluator; not a secure sandbox or hidden-test evaluator."
}

function Invoke-Git {
    param(
        [string]$WorkingDirectory,
        [string[]]$GitArguments
    )

    $output = & git -C $WorkingDirectory @GitArguments
    if ($LASTEXITCODE -ne 0) {
        throw "Git command failed: git -C `"$WorkingDirectory`" $($GitArguments -join ' ')"
    }

    return $output
}

function Get-GitStatus {
    param([string]$WorkingDirectory)

    $status = @(Invoke-Git $WorkingDirectory @(
        "status", "--porcelain", "--untracked-files=all"
    ))

    return ($status -join "`n").Trim()
}

function Get-ChangedPaths {
    param([string]$WorkingDirectory)

    $tracked = @(Invoke-Git $WorkingDirectory @(
        "diff", "--name-only"
    ))

    $staged = @(Invoke-Git $WorkingDirectory @(
        "diff", "--cached", "--name-only"
    ))

    $untracked = @(Invoke-Git $WorkingDirectory @(
        "ls-files", "--others", "--exclude-standard"
    ))

    return @(
        @($tracked) + @($staged) + @($untracked) |
            Where-Object { -not [string]::IsNullOrWhiteSpace($_) } |
            ForEach-Object { $_.Replace('\', '/') } |
            Sort-Object -Unique
    )
}


function Invoke-Pytest {
    param(
        [string]$WorkingDirectory,
        [string[]]$TestArguments = @()
    )

    Push-Location $WorkingDirectory
    try {
        $output = & python -m pytest @TestArguments 2>&1
        $exitCode = $LASTEXITCODE

        return [PSCustomObject]@{
            exit_code = $exitCode
            output = ($output | Out-String).Trim()
        }
    }
    finally {
        Pop-Location
    }
}

function Save-RunResult {
    New-Item -ItemType Directory -Path $RunsPath -Force |
        Out-Null

    $resultPath = Join-Path $RunsPath "P0-01_$RunId.json"

    [PSCustomObject]$Result |
        ConvertTo-Json -Depth 10 |
        Set-Content -LiteralPath $resultPath -Encoding UTF8

    Write-Host "Result file: $resultPath"
}

try {
    # Validate required files and worktree.
    if (-not (Test-Path $ManifestPath -PathType Leaf)) {
        throw "Task manifest not found: $ManifestPath"
    }

    if (-not (Test-Path $ResetScript -PathType Leaf)) {
        throw "Reset script not found: $ResetScript"
    }

    if (-not (Test-Path $WorktreePath -PathType Container)) {
        throw "Evaluation worktree not found: $WorktreePath"
    }

    $ResolvedPatch = (Resolve-Path -LiteralPath $PatchPath).Path

    if (-not (Test-Path $ResolvedPatch -PathType Leaf)) {
        throw "Patch file not found: $PatchPath"
    }

    $Result.patch_path = $ResolvedPatch

    # Require candidate patches to live outside the main repository.
    $mainPrefix = [System.IO.Path]::GetFullPath($MainRepoRoot).TrimEnd('\') + '\'
    $patchFullPath = [System.IO.Path]::GetFullPath($ResolvedPatch)

    if ($patchFullPath.StartsWith(
        $mainPrefix,
        [System.StringComparison]::OrdinalIgnoreCase
    )) {
        throw "Store the candidate patch outside mock_python_repo."
    }

    # Read the authoritative P0-01 task definition.
    $manifest = Get-Content $ManifestPath -Raw | ConvertFrom-Json
    $tasks = @($manifest.tasks | Where-Object { $_.task_id -eq "P0-01" })

    if ($tasks.Count -ne 1) {
        throw "Expected exactly one P0-01 task in tasks.json."
    }

    $task = $tasks[0]
    $targetCommit = [string]$task.base_revision

    if ($targetCommit -notmatch '^[0-9a-fA-F]{40}$') {
        throw "P0-01 base_revision must be a full 40-character commit hash."
    }

    # Verify the main repository is clean before resetting anything.
    $mainStatus = Get-GitStatus $MainRepoRoot
    if ($mainStatus) {
        throw "Main repository is dirty. Commit or resolve changes first:`n$mainStatus"
    }

    # Refuse to discard any worktree changes.
    $worktreeStatus = Get-GitStatus $WorktreePath
    if ($worktreeStatus) {
        throw "Evaluation worktree is dirty. No changes were removed:`n$worktreeStatus"
    }

    # Verify the worktree root is the expected directory.
    $actualRoot = (Invoke-Git $WorktreePath @(
        "rev-parse", "--show-toplevel"
    ) | Select-Object -First 1).Trim()

    $actualRoot = [System.IO.Path]::GetFullPath(
        $actualRoot.Replace('/', '\')
    ).TrimEnd('\')

    $expectedRoot = [System.IO.Path]::GetFullPath(
        $WorktreePath
    ).TrimEnd('\')

    if (-not $actualRoot.Equals(
        $expectedRoot,
        [System.StringComparison]::OrdinalIgnoreCase
    )) {
        throw "Unexpected evaluation worktree root."
    }

    # Reset only through the existing safety-checked reset script.
    & $ResetScript

    $startingCommit = (Invoke-Git $WorktreePath @(
        "rev-parse", "HEAD"
    ) | Select-Object -First 1).Trim()

    if ($startingCommit -ne $targetCommit) {
        throw "Starting revision differs from the manifest revision."
    }

    $Result.starting_revision = $startingCommit

    # Establish whether the defect is reproducible before evaluating a patch.
    Write-Host "`nRunning targeted P0-01 baseline tests..."

    $baseline = Invoke-Pytest $WorktreePath @(
        "tests/test_discounts.py", "-q"
    )

    $Result.baseline = $baseline
    $Result.diagnostics.baseline_tests_passed = ($baseline.exit_code -eq 0)
    Write-Host $baseline.output

    # Require the two known boundary failures and the observed test counts.
    $expectedFailures = @(
        "test_discount_rate_boundaries\[100-0\.05\]",
        "test_discounted_total"
    ) 

    $failureSignatureMatches = @(
        $expectedFailures | Where-Object {
            $baseline.output -match $_
        }
    ).Count -eq $expectedFailures.Count

    $expectedSummaryMatches = (
        $baseline.output -match "2 failed, 5 passed"
    )

    if (
        $baseline.exit_code -eq 0 -or
        -not $failureSignatureMatches -or
        -not $expectedSummaryMatches
    ) {
        throw "Baseline signature mismatch. Expected exactly the known P0-01 test outcome (2 failures, 5 passes). Candidate patch was not applied."
    }
        # Record the full-suite outcome before applying the candidate patch.
    Write-Host "`nRunning full-suite faulty-baseline tests..."

    $fullBaseline = Invoke-Pytest $WorktreePath
    $Result.full_suite_baseline = $fullBaseline
    $Result.diagnostics.full_suite_baseline_passed = (
        $fullBaseline.exit_code -eq 0
    )

    Write-Host $fullBaseline.output

    if (
        $fullBaseline.exit_code -eq 0 -or
